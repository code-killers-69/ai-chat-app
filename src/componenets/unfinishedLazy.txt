//  自定义指令配合 交叉观察器实现 懒加载

const observerImage = new IntersectionObserver(
    (entries) => {
        entries.forEach(entry => {
            if (entry.isIntersecting) {
                const img = entry.target.querySelector(".imageBlock")
                img.src = img.dataset.url
                observerImage.unobserve(entry.target)
            }
        });
    },
    {
        root: null,
        threshold: 0.25,
        rootMargin: "100px 0px 0px 0px",
    }
);

const vObserve = {
    mounted: (el) => {
        if (observerImage) {
            observerImage.observe(el)
        }
    },
    beforeMount(el) {
        const img = el.querySelector(".imageBlock")
        img.dataset.url = img.src
        img.src = ''
    },
    unmounted: (el) => {
        observerImage.unobserve(el)
    },
}
